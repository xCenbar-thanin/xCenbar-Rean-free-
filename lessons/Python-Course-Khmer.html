print() បង្ហាញលទ្ធផល។ # ជា comment មួយបន្ទាត់។ Python មិនប្រើ ; ឬ { } ទេ។

Python
print("Hello, World!")
print("Hello", "Python", sep=", ")   # sep = សញ្ញាខណ្ឌ

# នេះជា comment មួយបន្ទាត់

def greet():
    """នេះជា docstring៖ ពិពណ៌នាថា function ធ្វើអ្វី"""
    print("Hi")

print(greet.__doc__)    # អាន docstring
⚠️ ចំណាំសំខាន់
  • Python ញែកអក្សរតូចធំ៖ Print ≠ print។
  • ប្រើ quote "..." ឬ '...' បានទាំងពីរ។
  • Python 2 ឈប់គាំទ្រតាំងពីឆ្នាំ 2020 — ប្រើ Python 3 ជានិច្ច។
  • String បីសញ្ញា """...""" មិនមែនជា comment ពិតទេ — វាក្លាយជា docstring តែពេលនៅបន្ទាត់ដំបូងនៃ function, class ឬ module។

មេរៀន ២អថេរ និងប្រភេទទិន្នន័យ

មិនចាំបាច់ប្រកាសប្រភេទ Python កំណត់ដោយស្វ័យប្រវត្តិ (dynamic typing)។ ប្រភេទសំខាន់ៗ៖ int, float, str, bool, None។

Python
age = 20
gpa = 3.75
name = "Dara"
passed = True
nothing = None

print(type(age), type(gpa), type(name), type(passed))

a, b = 1, 2
a, b = b, a            # swap មួយបន្ទាត់
print(a, b)            # 2 1

print(int("5") + 1)    # 6   (str -> int)
print(float("2.5"))    # 2.5
print(str(10) + "kg")  # 10kg

big = 2 ** 100         # int មិនមានដែនកំណត់
print(big)
⚠️ ចំណាំសំខាន់
  • ឈ្មោះអថេរប្រើ snake_case (student_name) មិនចាប់ផ្តើមដោយលេខ មិនដូចពាក្យបម្រុង (if, for ។ល។)។
  • int ក្នុង Python មិនមាន overflow ដូច C/C++។
  • ប្រើអថេរមុនដាក់តម្លៃ នឹងកើត NameError។

    if n < 2:
        return False

def fib(n):                  # recursion តាមលំហាត់
    if n < 0:
        raise ValueError("n must be >= 0")
    if n < 2:                # base case

class Account:
    bank = "ABC Bank"            # class attribute

    def __init__(self, owner, balance=0):
        self.owner = owner
        self._balance = balance  # _ សំគាល់ថា private តាមប្រពៃណី

        if amount <= 0:

        if amount > self._balance:
            raise ValueError("Insufficient funds")
        self._balance -= amount

    def balance(self):
        return self._balance

    def __str__(self):
        return f"{self.owner}: ${self._balance}"

មេរៀន ១៨Generator, Decorator និង Context Manager

Generator (yield) បង្កើតតម្លៃម្តងមួយៗ សន្សំ memory។ Decorator បន្ថែមមុខងារទៅអនុគមន៍ដោយមិនកែកូដវា។ Context manager គ្រប់គ្រងការបើក/បិទ ក្នុង with។

def countdown(n):
    while n > 0:
        yield n           # ផ្អាក ហើយប្រគល់តម្លៃ
        n -= 1

for x in countdown(3):
    print(x)              # 3 2 1

print(sum(squares))                   # 30

import time, functools

def timer(func):
    @functools.wraps(func)
    def wrapper(*args, **kwargs):
        start = time.perf_counter()
        result = func(*args, **kwargs)
        print(f"{func.__name__} took {time.perf_counter() - start:.4f}s")
        return result
    return wrapper

@timer
def slow():
    return sum(range(1_000_000))

print(slow())

from contextlib import contextmanager

@contextmanager
def section(name):
    print("start", name)

        yield
    finally:

    data = request.get_json(silent=True)

        return jsonify({"error": "JSON object required"}), 400

    title = data.get("title")
    if not isinstance(title, str) or not title.strip():

    todos.append(todo)
    return jsonify(todo), 201

@app.get("/todos/<int:tid>")

    for t in todos:

    return jsonify({"error": "not found"}), 404

    if db is not None:
        db.close()

    def __init__(self, message: str, status: int = 400, details: dict | None = None):
        super().__init__(message)
        self.message = message
        self.status = status
        self.details = details or {}


def register_error_handlers(app: Flask) -> None:
    @app.errorhandler(AppError)
    def handle_app_error(err: AppError):
        body = {"error": err.message}
        if err.details:
            body["details"] = err.details
        return jsonify(body), err.status

    @app.errorhandler(HTTPException)
    def handle_http_error(err: HTTPException):
        return jsonify({"error": err.description}), err.code

    @app.errorhandler(Exception)
    def handle_unexpected(err: Exception):
        log.exception("Unexpected error")          # កត់ត្រា traceback ពេញ
        return jsonify({"error": "Internal server error"}), 500

import re

from .errors import AppError

EMAIL_RE = re.compile(r"^[\w.+-]+@[\w-]+(\.[\w-]+)+$")


def validate_student(data: object, partial: bool = False) -> dict:
    """ពិនិត្យ និងសម្អាតទិន្នន័យ; partial=True សម្រាប់ PATCH"""
    if not isinstance(data, dict):
        raise AppError("JSON object required")

    errors: dict[str, str] = {}
    clean: dict = {}

    if "name" in data or not partial:
        name = data.get("name")
        if not isinstance(name, str) or not name.strip():
            errors["name"] = "required, non-empty string"
        else:
            clean["name"] = name.strip()

    if "email" in data or not partial:
        email = data.get("email")
        if not isinstance(email, str) or not EMAIL_RE.match(email.strip()):
            errors["email"] = "valid email required"
        else:
            clean["email"] = email.strip().lower()

    if "score" in data:
        score = data["score"]
        if isinstance(score, bool) or not isinstance(score, (int, float)) or not 0 <= score <= 100:
            errors["score"] = "number between 0 and 100"
        else:
            clean["score"] = float(score)

    if errors:
        raise AppError("Validation failed", 422, errors)
    if partial and not clean:
        raise AppError("No valid fields to update")
    return clean

FIELDS = "id, name, email, score, created_at"


def list_students(search: str = "", page: int = 1, limit: int = 10) -> dict:
    db = get_db()
    like = f"%{search}%"
    total = db.execute(
        "SELECT COUNT(*) FROM students WHERE name LIKE ?", (like,)
    ).fetchone()[0]
    rows = db.execute(
        f"SELECT {FIELDS} FROM students WHERE name LIKE ? "
        "ORDER BY id LIMIT ? OFFSET ?",
        (like, limit, (page - 1) * limit),
    ).fetchall()
    return {
        "data": [dict(r) for r in rows],
        "total": total,
        "page": page,
        "pages": (total + limit - 1) // limit,
    }


def get_student(student_id: int) -> dict:
    row = get_db().execute(
        f"SELECT {FIELDS} FROM students WHERE id = ?", (student_id,)
    ).fetchone()
    if row is None:
        raise AppError("Student not found", 404)
    return dict(row)


def create_student(data: dict) -> dict:
    db = get_db()
    try:
        cur = db.execute(
            "INSERT INTO students (name, email, score) VALUES (?, ?, ?)",
            (data["name"], data["email"], data.get("score", 0)),
        )
        db.commit()
    except sqlite3.IntegrityError:
        raise AppError("Email already exists", 409)
    return get_student(cur.lastrowid)


def update_student(student_id: int, data: dict) -> dict:
    get_student(student_id)  # 404 បើមិនមាន
    # ឈ្មោះ column មកពី validator (មិនមែនពី user ផ្ទាល់) ដូច្នេះមានសុវត្ថិភាព
    sets = ", ".join(f"{col} = ?" for col in data)
    db = get_db()
    try:
        db.execute(
            f"UPDATE students SET {sets} WHERE id = ?", (*data.values(), student_id)
        )
        db.commit()
    except sqlite3.IntegrityError:
        raise AppError("Email already exists", 409)
    return get_student(student_id)


def delete_student(student_id: int) -> None:
    db = get_db()
    cur = db.execute("DELETE FROM students WHERE id = ?", (student_id,))
    db.commit()
    if cur.rowcount == 0:
        raise AppError("Student not found", 404)

class StudentApiTest(unittest.TestCase):

    def create(self, **overrides):
        body = {"name": "Dara", "email": "dara@mail.com", "score": 88}
        body.update(overrides)
        return self.client.post("/api/v1/students", json=body)

        self.assertEqual(res.status_code, 201)

def list_students(search="", page=1, limit=10, sort="id", order="asc"):
    if sort not in ALLOWED_SORT:            # Whitelist – មិនទុកចិត្ត Input

    ).fetchall()
