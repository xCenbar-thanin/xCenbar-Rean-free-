១. C# ជាអ្វី? អ្នកណាបង្កើត?

C# (អានថា “ស៊ី-សាប”) ជាភាសាសរសេរកម្មវិធីទំនើប ប្រភេទ object-oriented ដែលសុវត្ថិភាពប្រភេទទិន្នន័យ (type-safe) និងងាយរៀន។ កូដ C# ត្រូវបាន compile ហើយដំណើរការលើ .NET ដែលជាវេទិការបស់ Microsoft ដែល open-source និងដំណើរការបានលើ Windows, macOS និង Linux។

អ្នកបង្កើត

Anders Hejlsberg ជាស្ថាបត្យករចម្បង រួមជាមួយក្រុមវិស្វករ Microsoft។ គាត់ក៏ជាអ្នកបង្កើត Turbo Pascal និង Delphi ហើយក្រោយមកបង្កើត TypeScript ផងដែរ។

ក្រុមហ៊ុន

Microsoft ចេញផ្សាយជាផ្លូវការឆ្នាំ 2002 ជាមួយ .NET Framework ដំបូង (ឈ្មោះកូដដំបូង “Cool”)។

ស្តង់ដារ

ត្រូវបានកំណត់ជាស្តង់ដារដោយ ECMA និង ISO។ បច្ចុប្បន្នអភិវឌ្ឍជា open-source នៅលើ GitHub។

កំណែថ្មី

C# ចេញកំណែថ្មីរៀងរាល់ឆ្នាំជាមួយ .NET។ ឯកសារនេះប្រើ .NET 10 (LTS) និង C# 14។ កូដមូលដ្ឋានដំណើរការលើកំណែថ្មីៗទាំងអស់។

📌 ហេតុអ្វីឈ្មោះ “C Sharp”?
និមិត្តសញ្ញា ♯ មានន័យថា “ខ្ពស់ជាងមួយកម្រិត” ក្នុងតន្ត្រី ជាសញ្ញាបង្ហាញថាវាជាកំណែវិវឌ្ឍពី C/C++។ សូមកត់សម្គាល់៖ វាយក្នុងកុំព្យូទ័រជា C#។

៤. ដំឡើង C# ក្នុង VS Code

C# មិនមាន “កម្មវិធីដំឡើង” ដាច់ដោយឡែកទេ។ អ្នកត្រូវការ ២ យ៉ាង៖ .NET SDK (ជា compiler និងឧបករណ៍ដំណើរការ) និង VS Code ជាកម្មវិធីសរសេរកូដ ព្រមទាំង extension C#។

  1. ទាញយក និងដំឡើង .NET SDK
    ចូល dotnet.microsoft.com/download ជ្រើស .NET 10 SDK (ជំនាន់ LTS) តាមប្រព័ន្ធប្រតិបត្តិការរបស់អ្នក ហើយដំឡើងតាមលំនាំដើម។ សូមកុំជ្រើស “Runtime” ព្រោះអ្នកត្រូវការ SDK។
  2. ដំឡើង Visual Studio Code
    ទាញយកពី code.visualstudio.com ហើយដំឡើង។ លើ Windows សូមធីក “Add to PATH”។
  3. ដំឡើង Extension “C# Dev Kit”
    បើក VS Code ចុចរូប Extensions (Ctrl+Shift+X) ស្វែងរក C# Dev Kit (ដោយ Microsoft) ចុច Install។ វានឹងដំឡើង extension C# និង .NET Install Tool មកជាមួយ។

{

class BankAccount
{
    public string Owner { get; }
    public decimal Balance { get; private set; }

    public BankAccount(string owner, decimal initial)
    {
        Owner = owner;
        Balance = initial;
    }

    public void Deposit(decimal amount)
    {
        if (amount > 0) Balance += amount;
    }

    public bool Withdraw(decimal amount)
    {
        if (amount <= 0 || amount > Balance)
        {
            Console.WriteLine("សមតុល្យមិនគ្រប់គ្រាន់");
            return false;
        }
        Balance -= amount;
        return true;
    }
}

Generics អនុញ្ញាតឱ្យសរសេរកូដម្ដងប្រើបានជាមួយប្រភេទច្រើន ដោយរក្សា type-safety។ List<int> និង Dictionary<K,V> ជា generic ដែលអ្នកបានប្រើរួចហើយ។

class SimpleStack<T>
{
    private readonly List<T> _items = new();
    public void Push(T item) => _items.Add(item);
    public T Pop()
    {
        if (_items.Count == 0) throw new InvalidOperationException("ទទេ");
        var item = _items[^1];
        _items.RemoveAt(_items.Count - 1);
        return item;
    }
}

using UnityEngine;

public class PlayerMove : MonoBehaviour
{
    public float speed = 5f;       // បង្ហាញក្នុង Inspector ឱ្យកែបាន
    private int score = 0;

    void Start()
    {
        // ដំណើរការម្ដង ពេលហ្គេមចាប់ផ្ដើម
        Debug.Log("ហ្គេមចាប់ផ្ដើម!");
    }

    void Update()
    {
        // ដំណើរការរាល់ frame
        float h = Input.GetAxis("Horizontal");   // A/D ឬ ព្រួញឆ្វេង/ស្ដាំ
        float v = Input.GetAxis("Vertical");     // W/S ឬ ព្រួញឡើង/ចុះ
        Vector3 move = new Vector3(h, 0, v);

        transform.Translate(move * speed * Time.deltaTime);

        if (Input.GetKeyDown(KeyCode.Space))
        {
            if (TryGetComponent(out Rigidbody rb))      // កុំប្រើ ?. ជាមួយ Component របស់ Unity
                rb.AddForce(Vector3.up * 5f, ForceMode.Impulse);
        }
    }

    void OnTriggerEnter(Collider other)
    {
        if (other.CompareTag("Coin"))
        {
            score++;
            Debug.Log($"ពិន្ទុ: {score}");
            Destroy(other.gameObject);   // លុបកាក់
        }
    }
}

C#
const string FilePath = "todo.txt";
var tasks = Load();

while (true)
{
    Console.WriteLine("\n=== TODO ===");
    Console.WriteLine("1) បង្ហាញ  2) បន្ថែម  3) ធ្វើរួច  4) លុប  0) ចេញ");
    Console.Write("ជ្រើសរើស: ");

    switch (Console.ReadLine())
    {
        case "1": Show(); break;
        case "2":
            Console.Write("កិច្ចការ: ");
            var title = Console.ReadLine()?.Trim();
            if (!string.IsNullOrEmpty(title)) tasks.Add(new TodoItem(title, false));
            Save();
            break;
        case "3":
            Show();
            if (TryPick(out int d)) { tasks[d] = tasks[d] with { Done = true }; Save(); }
            break;
        case "4":
            Show();
            if (TryPick(out int r)) { tasks.RemoveAt(r); Save(); }
            break;
        case "0": return;
        default: Console.WriteLine("ជម្រើសមិនត្រឹមត្រូវ"); break;
    }
}

void Show()
{
    if (tasks.Count == 0) { Console.WriteLine("(គ្មានកិច្ចការ)"); return; }
    for (int i = 0; i < tasks.Count; i++)
        Console.WriteLine($"{i + 1}. [{(tasks[i].Done ? "x" : " ")}] {tasks[i].Title}");
    Console.WriteLine($"ធ្វើរួច {tasks.Count(t => t.Done)}/{tasks.Count}");
}

bool TryPick(out int index)
{
    Console.Write("លេខរៀង: ");
    bool ok = int.TryParse(Console.ReadLine(), out index);
    index--;
    return ok && index >= 0 && index < tasks.Count;
}

List<TodoItem> Load() =>
    !File.Exists(FilePath) ? new() :
    File.ReadAllLines(FilePath)
        .Select(l => l.Split('|', 2))   // 2 = បំបែកតែលើកដំបូង (ចំណងជើងអាចមាន '|')
        .Where(p => p.Length == 2)
        .Select(p => new TodoItem(p[1], p[0] == "1"))
        .ToList();

void Save() =>
    File.WriteAllLines(FilePath, tasks.Select(t => $"{(t.Done ? 1 : 0)}|{t.Title}"));

record TodoItem(string Title, bool Done);

using Microsoft.EntityFrameworkCore;

var builder = WebApplication.CreateBuilder(args);
builder.Services.AddDbContext<TodoDb>(o => o.UseSqlite("Data Source=todo.db"));
var app = builder.Build();

using (var scope = app.Services.CreateScope())
    scope.ServiceProvider.GetRequiredService<TodoDb>().Database.EnsureCreated();

app.MapGet("/todos", async (TodoDb db) => await db.Todos.ToListAsync());

app.MapGet("/todos/{id:int}", async (int id, TodoDb db) =>
    await db.Todos.FindAsync(id) is Todo t ? Results.Ok(t) : Results.NotFound());

app.MapPost("/todos", async (Todo todo, TodoDb db) =>
{
    db.Todos.Add(todo);
    await db.SaveChangesAsync();
    return Results.Created($"/todos/{todo.Id}", todo);
});

app.MapPut("/todos/{id:int}", async (int id, Todo input, TodoDb db) =>
{
    var todo = await db.Todos.FindAsync(id);
    if (todo is null) return Results.NotFound();
    todo.Title = input.Title;
    todo.Done = input.Done;
    await db.SaveChangesAsync();
    return Results.NoContent();
});

app.MapDelete("/todos/{id:int}", async (int id, TodoDb db) =>
{
    var todo = await db.Todos.FindAsync(id);
    if (todo is null) return Results.NotFound();
    db.Todos.Remove(todo);
    await db.SaveChangesAsync();
    return Results.NoContent();
});

app.Run();

class Todo
{
    public int Id { get; set; }
    public string Title { get; set; } = "";
    public bool Done { get; set; }
}

class TodoDb(DbContextOptions<TodoDb> options) : DbContext(options)
{
    public DbSet<Todo> Todos => Set<Todo>();
}

static int Search(int[] arr, int target)
{
    int lo = 0, hi = arr.Length - 1;
    while (lo <= hi)
    {
        int mid = lo + (hi - lo) / 2;
        if (arr[mid] == target) return mid;
        if (arr[mid] < target) lo = mid + 1;
        else hi = mid - 1;
    }
    return -1;
}

គ២. ដង្កៀបត្រឹមត្រូវ

ពិនិត្យ string ដែលមាន ()[]{} ថាបិទបើកត្រឹមត្រូវឬទេ ដោយប្រើ Stack។

static bool IsBalanced(string s)
{
    var pairs = new Dictionary<char, char> { [')'] = '(', [']'] = '[', ['}'] = '{' };
    var stack = new Stack<char>();
    foreach (char c in s)
    {
        if ("([{".Contains(c)) stack.Push(c);
        else if (pairs.TryGetValue(c, out char open))
        {
            if (stack.Count == 0 || stack.Pop() != open) return false;
        }
    }
    return stack.Count == 0;
}

class Book(string title) { public string Title { get; } = title; public bool Borrowed { get; set; } }

class Library
{
    private readonly List<Book> _books = new();
    public void Add(string title) => _books.Add(new Book(title));

    public bool Borrow(string title)
    {
        var b = _books.FirstOrDefault(x => x.Title == title && !x.Borrowed);
        if (b is null) return false;
        b.Borrowed = true;
        return true;
    }

    public void Return(string title)
    {
        var b = _books.FirstOrDefault(x => x.Title == title);
        if (b is not null) b.Borrowed = false;
    }

    public IEnumerable<string> Available() =>
        _books.Where(b => !b.Borrowed).Select(b => b.Title);
}

using Microsoft.EntityFrameworkCore;
using TodoApi.Data;
using TodoApi.Dtos;
using TodoApi.Models;

namespace TodoApi.Services;

public class TodoService(TodoDb db) : ITodoService
{
    public async Task<IReadOnlyList<TodoResponse>> ListAsync(bool? done, int page, int pageSize, CancellationToken ct)
    {
        IQueryable<Todo> query = db.Todos.AsNoTracking();
        if (done is bool d)
            query = query.Where(t => t.Done == d);

        return await query
            .OrderBy(t => t.Id)
            .Skip((page - 1) * pageSize)
            .Take(pageSize)
            .Select(t => new TodoResponse(t.Id, t.Title, t.Done, t.CreatedAt))
            .ToListAsync(ct);
    }

    public Task<TodoResponse?> GetAsync(int id, CancellationToken ct) =>
        db.Todos.AsNoTracking()
            .Where(t => t.Id == id)
            .Select(t => new TodoResponse(t.Id, t.Title, t.Done, t.CreatedAt))
            .FirstOrDefaultAsync(ct);

    public async Task<TodoResponse> CreateAsync(string title, CancellationToken ct)
    {
        var todo = new Todo { Title = title };
        db.Todos.Add(todo);
        await db.SaveChangesAsync(ct);
        return new TodoResponse(todo.Id, todo.Title, todo.Done, todo.CreatedAt);
    }

    public async Task<bool> UpdateAsync(int id, string title, bool done, CancellationToken ct)
    {
        var todo = await db.Todos.FindAsync(new object[] { id }, ct);
        if (todo is null) return false;
        todo.Title = title;
        todo.Done = done;
        await db.SaveChangesAsync(ct);
        return true;
    }

    public async Task<bool> DeleteAsync(int id, CancellationToken ct) =>
        await db.Todos.Where(t => t.Id == id).ExecuteDeleteAsync(ct) > 0;
}

TodoApi/Endpoints/TodoEndpoints.cs
using Microsoft.AspNetCore.Http.HttpResults;
using TodoApi.Dtos;
using TodoApi.Services;

namespace TodoApi.Endpoints;

public static class TodoEndpoints
{
    private const int MaxTitleLength = 200;

    public static void MapTodoEndpoints(this IEndpointRouteBuilder app)
    {
        var group = app.MapGroup("/api/todos").WithTags("Todos");

        // GET /api/todos?done=true&page=1&pageSize=20
        group.MapGet("/", async (ITodoService svc, bool? done, int? page, int? pageSize, CancellationToken ct) =>
        {
            int p = Math.Max(page ?? 1, 1);
            int size = Math.Clamp(pageSize ?? 20, 1, 100);
            return TypedResults.Ok(await svc.ListAsync(done, p, size, ct));
        });

        group.MapGet("/{id:int}", async Task<Results<Ok<TodoResponse>, NotFound>> (
            int id, ITodoService svc, CancellationToken ct) =>
        {
            var todo = await svc.GetAsync(id, ct);
            return todo is null ? TypedResults.NotFound() : TypedResults.Ok(todo);
        });

        group.MapPost("/", async Task<Results<Created<TodoResponse>, ValidationProblem>> (
            CreateTodoRequest req, ITodoService svc, CancellationToken ct) =>
        {
            if (ValidateTitle(req.Title) is { } errors)
                return TypedResults.ValidationProblem(errors);

            var todo = await svc.CreateAsync(req.Title!.Trim(), ct);
            return TypedResults.Created($"/api/todos/{todo.Id}", todo);
        });

        group.MapPut("/{id:int}", async Task<Results<NoContent, NotFound, ValidationProblem>> (
            int id, UpdateTodoRequest req, ITodoService svc, CancellationToken ct) =>
        {
            if (ValidateTitle(req.Title) is { } errors)
                return TypedResults.ValidationProblem(errors);

            bool updated = await svc.UpdateAsync(id, req.Title!.Trim(), req.Done, ct);
            return updated ? TypedResults.NoContent() : TypedResults.NotFound();
        });

        group.MapDelete("/{id:int}", async Task<Results<NoContent, NotFound>> (
            int id, ITodoService svc, CancellationToken ct) =>
        {
            bool deleted = await svc.DeleteAsync(id, ct);
            return deleted ? TypedResults.NoContent() : TypedResults.NotFound();
        });
    }

    // ត្រឡប់ null បើត្រឹមត្រូវ ឬ dictionary នៃកំហុស (→ 400 ValidationProblem)
    private static Dictionary<string, string[]>? ValidateTitle(string? title)
    {
        if (string.IsNullOrWhiteSpace(title))
            return new() { ["title"] = new[] { "Title is required." } };
        if (title.Trim().Length > MaxTitleLength)
            return new() { ["title"] = new[] { $"Title must be {MaxTitleLength} characters or fewer." } };
        return null;
    }
}

public class TodoApiTests : IDisposable
{
    private readonly WebApplicationFactory<Program> _factory;
    private readonly HttpClient _client;

    public TodoApiTests()
    {
        var dbPath = Path.Combine(Path.GetTempPath(), $"todo-test-{Guid.NewGuid()}.db");
        Environment.SetEnvironmentVariable("ConnectionStrings__Todo", $"Data Source={dbPath}");
        _factory = new WebApplicationFactory<Program>();
        _client = _factory.CreateClient();

    public void Dispose()
    {
        _client.Dispose();
        _factory.Dispose();
    }

    [Fact]
    public async Task Health_ReturnsOk()
    {
        var res = await _client.GetAsync("/health");
        Assert.Equal(HttpStatusCode.OK, res.StatusCode);
    }

    [Fact]
    public async Task Create_ThenGet_ReturnsSameTodo()
    {
        var created = await _client.PostAsJsonAsync("/api/todos", new CreateTodoRequest("Learn C#"));
        Assert.Equal(HttpStatusCode.Created, created.StatusCode);

        var todo = await created.Content.ReadFromJsonAsync<TodoResponse>();
        Assert.NotNull(todo);

        var fetched = await _client.GetFromJsonAsync<TodoResponse>($"/api/todos/{todo.Id}");
        Assert.Equal("Learn C#", fetched!.Title);
        Assert.False(fetched.Done);
    }

    [Theory]
    [InlineData("")]
    [InlineData("   ")]
    public async Task Create_WithEmptyTitle_ReturnsBadRequest(string title)
    {
        var res = await _client.PostAsJsonAsync("/api/todos", new CreateTodoRequest(title));
        Assert.Equal(HttpStatusCode.BadRequest, res.StatusCode);
    }

    [Fact]
    public async Task Update_Filter_Delete_Flow()
    {
        var created = await _client.PostAsJsonAsync("/api/todos", new CreateTodoRequest("Write tests"));
        var todo = (await created.Content.ReadFromJsonAsync<TodoResponse>())!;

        var put = await _client.PutAsJsonAsync($"/api/todos/{todo.Id}", new UpdateTodoRequest("Write tests", true));
        Assert.Equal(HttpStatusCode.NoContent, put.StatusCode);

        var doneList = await _client.GetFromJsonAsync<List<TodoResponse>>("/api/todos?done=true");
        Assert.Single(doneList!);

        var del = await _client.DeleteAsync($"/api/todos/{todo.Id}");
        Assert.Equal(HttpStatusCode.NoContent, del.StatusCode);

        var missing = await _client.GetAsync($"/api/todos/{todo.Id}");
        Assert.Equal(HttpStatusCode.NotFound, missing.StatusCode);
    }
