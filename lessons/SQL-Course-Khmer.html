១. SQL ជាអ្វី?

SQL (Structured Query Language — អានថា «អេស-គីវ-អែល» ឬ «ស៊ីឃ្វែល») គឺជា ភាសាសម្រាប់គ្រប់គ្រង និងសួរទិន្នន័យ ក្នុង Relational Database (មូលដ្ឋានទិន្នន័យដែលរក្សាទុកទិន្នន័យជាតារាង មានជួរដេក និងជួរឈរ)។

ជាមួយ SQL យើងអាច៖ បង្កើតតារាង បញ្ចូល អាន កែប្រែ លុបទិន្នន័យ ហើយធ្វើរបាយការណ៍ពីទិន្នន័យរាប់លានជួរក្នុងរយៈពេលប៉ុន្មានវិនាទី។

-- បង្កើត Database (utf8mb4 គាំទ្រអក្សរខ្មែរ និង Emoji)
CREATE DATABASE IF NOT EXISTS shop_db
  CHARACTER SET utf8mb4
  COLLATE utf8mb4_unicode_ci;

-- ជ្រើសប្រើ Database
USE shop_db;

-- បង្ហាញ Database ទាំងអស់
SHOW DATABASES;

-- តារាង categories
CREATE TABLE categories (
  id    INT AUTO_INCREMENT PRIMARY KEY,
  name  VARCHAR(100) NOT NULL UNIQUE
);

-- តារាង products
CREATE TABLE products (
  id           INT AUTO_INCREMENT PRIMARY KEY,
  category_id  INT,
  name         VARCHAR(150) NOT NULL,
  price        DECIMAL(10, 2) NOT NULL CHECK (price >= 0),
  stock        INT NOT NULL DEFAULT 0,
  created_at   TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (category_id) REFERENCES categories(id)
);

-- តារាង customers
CREATE TABLE customers (
  id     INT AUTO_INCREMENT PRIMARY KEY,
  name   VARCHAR(100) NOT NULL,
  email  VARCHAR(150) UNIQUE,
  city   VARCHAR(50)
);

-- តារាង orders
CREATE TABLE orders (
  id           INT AUTO_INCREMENT PRIMARY KEY,
  customer_id  INT NOT NULL,
  order_date   DATE NOT NULL,
  status       VARCHAR(20) DEFAULT 'pending',
  FOREIGN KEY (customer_id) REFERENCES customers(id)
);

-- តារាង order_items
CREATE TABLE order_items (
  id          INT AUTO_INCREMENT PRIMARY KEY,
  order_id    INT NOT NULL,
  product_id  INT NOT NULL,
  quantity    INT NOT NULL CHECK (quantity > 0),
  price       DECIMAL(10, 2) NOT NULL,
  FOREIGN KEY (order_id) REFERENCES orders(id) ON DELETE CASCADE,
  FOREIGN KEY (product_id) REFERENCES products(id)
);

SHOW TABLES;
DESCRIBE products;

-- Aggregate functions
SELECT
  COUNT(*)            AS total_products,
  SUM(stock)          AS total_stock,
  AVG(price)          AS avg_price,
  MIN(price)          AS cheapest,
  MAX(price)          AS most_expensive
FROM products;

FROM customers
GROUP BY city
ORDER BY total_customers DESC;

FROM products
GROUP BY category_id;

FROM orders
WHERE status = 'completed'
GROUP BY customer_id
HAVING COUNT(*) >= 2;

-- INNER JOIN: ផលិតផល + ឈ្មោះប្រភេទ
SELECT p.name AS product, c.name AS category, p.price
FROM products p
INNER JOIN categories c ON p.category_id = c.id;

-- ការកុម្ម៉ង់ + ឈ្មោះអតិថិជន
SELECT o.id AS order_id, cu.name AS customer, o.order_date, o.status
FROM orders o
JOIN customers cu ON o.customer_id = cu.id;

-- LEFT JOIN: អតិថិជនទាំងអស់ រួមទាំងអ្នកមិនដែលកុម្ម៉ង់
SELECT cu.name, o.id AS order_id
FROM customers cu
LEFT JOIN orders o ON o.customer_id = cu.id;

-- រកអតិថិជនដែលមិនដែលកុម្ម៉ង់
SELECT cu.name
FROM customers cu
LEFT JOIN orders o ON o.customer_id = cu.id
WHERE o.id IS NULL;

-- JOIN ច្រើនតារាង: វិក្កយបត្រលម្អិត
SELECT
  o.id                     AS order_id,
  cu.name                  AS customer,
  p.name                   AS product,
  oi.quantity,
  oi.price,
  oi.quantity * oi.price   AS line_total
FROM orders o
JOIN customers   cu ON cu.id = o.customer_id
JOIN order_items oi ON oi.order_id = o.id
JOIN products    p  ON p.id = oi.product_id
ORDER BY o.id;

-- SELF JOIN: បុគ្គលិក + ឈ្មោះអ្នកគ្រប់គ្រង
SELECT e.name AS employee, m.name AS manager
FROM employees e
LEFT JOIN employees m ON e.manager_id = m.id;

WITH order_totals AS (
  SELECT order_id, SUM(quantity * price) AS total
  FROM order_items
  GROUP BY order_id
),
customer_totals AS (
  SELECT o.customer_id, SUM(ot.total) AS spent
  FROM orders o
  JOIN order_totals ot ON ot.order_id = o.id
  GROUP BY o.customer_id
)
SELECT cu.name, ct.spent
FROM customer_totals ct
JOIN customers cu ON cu.id = ct.customer_id
WHERE ct.spent > 10
ORDER BY ct.spent DESC;

-- Recursive CTE: បង្កើតលេខ 1 ដល់ 10
WITH RECURSIVE nums AS (
  SELECT 1 AS n
  UNION ALL
  SELECT n + 1 FROM nums WHERE n < 10
)
SELECT n FROM nums;

មេរៀន ១២Transactions & ACID

Transaction ធានាថា Query មួយក្រុម ជោគជ័យទាំងអស់ ឬ បរាជ័យទាំងអស់។ ឧ. ផ្ទេរប្រាក់: ដកពី A ហើយបូកឱ្យ B — មិនអាចធ្វើតែពាក់កណ្តាលបានទេ។

ACIDអត្ថន័យ
Atomicityទាំងអស់ ឬគ្មានអ្វីសោះ
Consistencyទិន្នន័យតែងតែត្រឹមត្រូវតាមច្បាប់ (Constraints)
IsolationTransaction នីមួយៗមិនរំខានគ្នា
Durabilityពេល COMMIT ហើយ ទិន្នន័យមិនបាត់ ទោះដាច់ភ្លើង

-- Stored Procedure: យកការកុម្ម៉ង់របស់អតិថិជន
CREATE PROCEDURE get_customer_orders(IN p_customer_id INT)
BEGIN
  SELECT o.id, o.order_date, o.status, SUM(oi.quantity * oi.price) AS total
  FROM orders o
  JOIN order_items oi ON oi.order_id = o.id
  WHERE o.customer_id = p_customer_id
  GROUP BY o.id, o.order_date, o.status;
END //

-- Function: គណនាតម្លៃក្រោយបញ្ចុះ
CREATE FUNCTION discounted_price(p_price DECIMAL(10,2), p_percent INT)
RETURNS DECIMAL(10,2)
DETERMINISTIC
BEGIN
  RETURN ROUND(p_price * (100 - p_percent) / 100, 2);
END //

DELIMITER ;

AFTER UPDATE ON products

  IF OLD.price <> NEW.price THEN

    VALUES (NEW.id, OLD.price, NEW.price);

Window Function គណនាលើក្រុមជួរ ដោយ មិនបង្រួមជួរ ដូច GROUP BY។ (MySQL 8+, PostgreSQL)

window.sql
-- ចំណាត់ថ្នាក់ផលិតផលតាមតម្លៃ ក្នុងប្រភេទនីមួយៗ
SELECT
  category_id, name, price,
  ROW_NUMBER() OVER (PARTITION BY category_id ORDER BY price DESC) AS row_num,
  RANK()       OVER (PARTITION BY category_id ORDER BY price DESC) AS rnk,
  DENSE_RANK() OVER (ORDER BY price DESC)                          AS overall_rank
FROM products;

-- តម្លៃមធ្យមនៃប្រភេទ បង្ហាញក្បែរជួរនីមួយៗ
SELECT name, category_id, price,
  ROUND(AVG(price) OVER (PARTITION BY category_id), 2) AS category_avg,
  price - AVG(price) OVER (PARTITION BY category_id)   AS diff
FROM products;

-- ចំណូលបូកបន្ត (Running total) តាមថ្ងៃ
WITH daily AS (
  SELECT o.order_date, SUM(oi.quantity * oi.price) AS revenue
  FROM orders o JOIN order_items oi ON oi.order_id = o.id
  GROUP BY o.order_date
)
SELECT order_date, revenue,
  SUM(revenue) OVER (ORDER BY order_date)        AS running_total,
  LAG(revenue)  OVER (ORDER BY order_date)       AS prev_day,
  revenue - LAG(revenue) OVER (ORDER BY order_date) AS change_amount
FROM daily;

-- Top 1 ផលិតផលថ្លៃបំផុតក្នុងប្រភេទនីមួយៗ
WITH ranked AS (
  SELECT *, ROW_NUMBER() OVER (PARTITION BY category_id ORDER BY price DESC) AS rn
  FROM products
)
SELECT category_id, name, price FROM ranked WHERE rn = 1;

WITH spending AS (
  SELECT cu.name, SUM(oi.quantity * oi.price) AS spent
  FROM customers cu
  JOIN orders o       ON o.customer_id = cu.id
  JOIN order_items oi ON oi.order_id = o.id

    WHEN 2 THEN 'Silver'
    ELSE 'Bronze'

  role_id        TINYINT UNSIGNED NOT NULL,
  full_name      VARCHAR(100) NOT NULL,

  CONSTRAINT fk_users_role FOREIGN KEY (role_id) REFERENCES roles(id)

CREATE TABLE categories (
  id          INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  parent_id   INT UNSIGNED NULL,                    -- ប្រភេទរង (Tree)
  name        VARCHAR(100) NOT NULL,
  slug        VARCHAR(120) NOT NULL UNIQUE,
  CONSTRAINT fk_categories_parent FOREIGN KEY (parent_id) REFERENCES categories(id)
) ENGINE = InnoDB;

CREATE TABLE products (
  id           BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  category_id  INT UNSIGNED NOT NULL,
  sku          VARCHAR(50) NOT NULL UNIQUE,          -- លេខកូដទំនិញ / Barcode
  name         VARCHAR(150) NOT NULL,
  description  TEXT,
  cost_price   DECIMAL(12,2) NOT NULL DEFAULT 0 CHECK (cost_price >= 0),
  sell_price   DECIMAL(12,2) NOT NULL CHECK (sell_price >= 0),
  stock        INT NOT NULL DEFAULT 0 CHECK (stock >= 0),
  reorder_level INT NOT NULL DEFAULT 10,
  created_at   DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  updated_at   DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  deleted_at   DATETIME NULL,                        -- Soft delete
  CONSTRAINT fk_products_category FOREIGN KEY (category_id) REFERENCES categories(id),
  INDEX idx_products_name (name),
  INDEX idx_products_category (category_id)
) ENGINE = InnoDB;

-- ===== Customers =====
CREATE TABLE customers (
  id          BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  full_name   VARCHAR(100) NOT NULL,
  phone       VARCHAR(20) UNIQUE,
  email       VARCHAR(150) UNIQUE,
  points      INT NOT NULL DEFAULT 0,               -- ពិន្ទុសមាជិក
  created_at  DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP

CREATE TABLE sales (
  id              BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  invoice_no      VARCHAR(30) NOT NULL UNIQUE,
  customer_id     BIGINT UNSIGNED NULL,
  cashier_id      BIGINT UNSIGNED NOT NULL,
  status          ENUM('draft', 'paid', 'refunded', 'void') NOT NULL DEFAULT 'draft',
  subtotal        DECIMAL(12,2) NOT NULL DEFAULT 0,
  discount        DECIMAL(12,2) NOT NULL DEFAULT 0,
  tax             DECIMAL(12,2) NOT NULL DEFAULT 0,
  total           DECIMAL(12,2) NOT NULL DEFAULT 0,
  payment_method  ENUM('cash', 'card', 'khqr') NOT NULL DEFAULT 'cash',
  sold_at         DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  CONSTRAINT fk_sales_customer FOREIGN KEY (customer_id) REFERENCES customers(id),
  CONSTRAINT fk_sales_cashier  FOREIGN KEY (cashier_id)  REFERENCES users(id),
  INDEX idx_sales_sold_at (sold_at),
  INDEX idx_sales_status_date (status, sold_at)
) ENGINE = InnoDB;

CREATE TABLE sale_items (
  id          BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  sale_id     BIGINT UNSIGNED NOT NULL,
  product_id  BIGINT UNSIGNED NOT NULL,
  quantity    INT NOT NULL CHECK (quantity > 0),
  unit_price  DECIMAL(12,2) NOT NULL,     -- តម្លៃពេលលក់ (មិនប្តូរតាមផលិតផល)
  unit_cost   DECIMAL(12,2) NOT NULL,     -- សម្រាប់គណនាប្រាក់ចំណេញ
  line_total  DECIMAL(12,2) AS (quantity * unit_price) STORED,  -- Generated column
  CONSTRAINT fk_items_sale    FOREIGN KEY (sale_id)    REFERENCES sales(id) ON DELETE CASCADE,
  CONSTRAINT fk_items_product FOREIGN KEY (product_id) REFERENCES products(id)
) ENGINE = InnoDB;

-- ===== Inventory movements (Audit ស្តុក) =====
CREATE TABLE stock_movements (
  id          BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  product_id  BIGINT UNSIGNED NOT NULL,
  change_qty  INT NOT NULL,                              -- +ចូល / -ចេញ
  reason      ENUM('purchase', 'sale', 'refund', 'adjust') NOT NULL,
  ref_id      BIGINT UNSIGNED NULL,                      -- sale_id ឬ purchase_id
  user_id     BIGINT UNSIGNED NULL,
  created_at  DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  CONSTRAINT fk_moves_product FOREIGN KEY (product_id) REFERENCES products(id),
  INDEX idx_moves_product_date (product_id, created_at)

Stored Procedure: ការលក់មួយ (Transaction + Error handling)

sp_add_sale_item.sql
DELIMITER //

CREATE PROCEDURE sp_add_sale_item(
  IN p_sale_id BIGINT UNSIGNED,
  IN p_product_id BIGINT UNSIGNED,
  IN p_qty INT,
  IN p_user_id BIGINT UNSIGNED
)
BEGIN
  DECLARE v_stock INT;
  DECLARE v_price DECIMAL(12,2);
  DECLARE v_cost  DECIMAL(12,2);

  DECLARE EXIT HANDLER FOR SQLEXCEPTION
  BEGIN
    ROLLBACK;
    RESIGNAL;   -- បញ្ជូន Error ទៅ App
  END;

  START TRANSACTION;

  -- FOR UPDATE: ចាក់សោជួរនេះ ការពារការលក់ក្នុងពេលតែមួយ (Race condition)
  SELECT stock, sell_price, cost_price INTO v_stock, v_price, v_cost
  FROM products
  WHERE id = p_product_id AND deleted_at IS NULL
  FOR UPDATE;

  IF v_stock IS NULL THEN
    SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'រកមិនឃើញផលិតផល';
  END IF;

  IF v_stock < p_qty THEN
    SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'ស្តុកមិនគ្រប់គ្រាន់';
  END IF;

  INSERT INTO sale_items (sale_id, product_id, quantity, unit_price, unit_cost)
  VALUES (p_sale_id, p_product_id, p_qty, v_price, v_cost);

  UPDATE products SET stock = stock - p_qty WHERE id = p_product_id;

  INSERT INTO stock_movements (product_id, change_qty, reason, ref_id, user_id)
  VALUES (p_product_id, -p_qty, 'sale', p_sale_id, p_user_id);

  UPDATE sales
  SET subtotal = (SELECT SUM(line_total) FROM sale_items WHERE sale_id = p_sale_id),
      total    = subtotal - discount + tax
  WHERE id = p_sale_id;

  COMMIT;
END //

DELIMITER ;

Query របាយការណ៍ពេញនិយម

reports.sql
-- 1) ចំណូល ថ្លៃដើម ប្រាក់ចំណេញ តាមថ្ងៃ (៣០ថ្ងៃចុងក្រោយ)
SELECT DATE(s.sold_at)                                   AS sale_date,
       COUNT(DISTINCT s.id)                              AS invoices,
       SUM(si.line_total)                                AS revenue,
       SUM(si.quantity * si.unit_cost)                   AS cost,
       SUM(si.line_total - si.quantity * si.unit_cost)   AS profit
FROM sales s
JOIN sale_items si ON si.sale_id = s.id
WHERE s.status = 'paid'
  AND s.sold_at >= CURDATE() - INTERVAL 30 DAY
GROUP BY DATE(s.sold_at)
ORDER BY sale_date;

-- 2) Top 10 ផលិតផលលក់ដាច់បំផុត ខែនេះ
SELECT p.sku, p.name, SUM(si.quantity) AS qty_sold, SUM(si.line_total) AS revenue
FROM sale_items si
JOIN sales s    ON s.id = si.sale_id
JOIN products p ON p.id = si.product_id
WHERE s.status = 'paid'
  AND s.sold_at >= DATE_FORMAT(CURDATE(), '%Y-%m-01')
GROUP BY p.id, p.sku, p.name
ORDER BY qty_sold DESC
LIMIT 10;

-- 3) ទំនិញដែលត្រូវបញ្ជាទិញបន្ថែម
SELECT sku, name, stock, reorder_level
FROM products
WHERE deleted_at IS NULL AND stock <= reorder_level
ORDER BY stock;

-- 4) ការលក់តាមអ្នកគិតលុយ និងវិធីបង់ប្រាក់
SELECT u.full_name AS cashier,
       SUM(CASE WHEN s.payment_method = 'cash' THEN s.total ELSE 0 END) AS cash,
       SUM(CASE WHEN s.payment_method = 'khqr' THEN s.total ELSE 0 END) AS khqr,
       SUM(CASE WHEN s.payment_method = 'card' THEN s.total ELSE 0 END) AS card,
       SUM(s.total) AS total
FROM sales s
JOIN users u ON u.id = s.cashier_id
WHERE s.status = 'paid' AND DATE(s.sold_at) = CURDATE()
GROUP BY u.id, u.full_name;

-- 5) កំណើនចំណូលធៀបខែមុន (Month-over-Month)
WITH monthly AS (
  SELECT DATE_FORMAT(sold_at, '%Y-%m') AS ym, SUM(total) AS revenue
  FROM sales WHERE status = 'paid'
  GROUP BY ym
)
SELECT ym, revenue,
       LAG(revenue) OVER (ORDER BY ym) AS prev_month,
       ROUND((revenue - LAG(revenue) OVER (ORDER BY ym))
             / LAG(revenue) OVER (ORDER BY ym) * 100, 1) AS growth_pct
FROM monthly;

CREATE PROCEDURE sp_refund_sale(IN p_sale_id BIGINT UNSIGNED, IN p_user_id BIGINT UNSIGNED)
BEGIN
  DECLARE v_status VARCHAR(20);
  DECLARE EXIT HANDLER FOR SQLEXCEPTION BEGIN ROLLBACK; RESIGNAL; END;

  START TRANSACTION;

  SELECT status INTO v_status FROM sales WHERE id = p_sale_id FOR UPDATE;

  IF v_status IS NULL OR v_status <> 'paid' THEN
    SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'អាច Refund បានតែការលក់ដែលបានបង់ប្រាក់';
  END IF;

  -- បូកស្តុកវិញ
  UPDATE products p
  JOIN sale_items si ON si.product_id = p.id
  SET p.stock = p.stock + si.quantity
  WHERE si.sale_id = p_sale_id;

  -- កត់ត្រាចលនាស្តុក
  INSERT INTO stock_movements (product_id, change_qty, reason, ref_id, user_id)
  SELECT product_id, quantity, 'refund', p_sale_id, p_user_id
  FROM sale_items WHERE sale_id = p_sale_id;

  UPDATE sales SET status = 'refunded' WHERE id = p_sale_id;

  COMMIT;
END //
