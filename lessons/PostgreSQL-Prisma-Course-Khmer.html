១. PostgreSQL និង Prisma ជាអ្វី? អ្នកណាបង្កើត?

🧱 Migration + place_order() (រត់ពិត)

លទ្ធផលពិត · PostgreSQL 16 (psql)
kafe=# CREATE INDEX IF NOT EXISTS orders_customer_idx ON orders (customer_id);
CREATE INDEX
kafe=# CREATE INDEX IF NOT EXISTS order_items_product_idx ON order_items (product_id);
CREATE INDEX
kafe=# CREATE OR REPLACE FUNCTION place_order(p_customer integer, p_items jsonb)
kafe-# RETURNS TABLE (order_id integer, total numeric) LANGUAGE plpgsql AS $$
kafe-# DECLARE
kafe-#   v_order integer;
kafe-#   it jsonb;
kafe-#   v_price numeric;
kafe-#   v_stock integer;
kafe-# BEGIN
kafe-#   IF jsonb_array_length(p_items) = 0 THEN RAISE EXCEPTION 'Order ត្រូវមានយ៉ាងតិច ១ មុខ'; END IF;
kafe-#   INSERT INTO orders (customer_id, status, created_at) VALUES (p_customer, 'paid', now()) RETURNING id INTO v_order;
kafe-#   FOR it IN SELECT * FROM jsonb_array_elements(p_items) LOOP
kafe-#     SELECT price, stock INTO v_price, v_stock FROM products
kafe-#      WHERE id = (it->>'product_id')::int FOR UPDATE;           -- Lock ជួរ
kafe-#     IF NOT FOUND THEN RAISE EXCEPTION 'រកមិនឃើញផលិតផល #%', it->>'product_id'; END IF;
kafe-#     IF v_stock < (it->>'qty')::int THEN
kafe-#       RAISE EXCEPTION 'ស្តុកមិនគ្រប់: #% នៅសល់ %', it->>'product_id', v_stock;
kafe-#     END IF;
kafe-#     UPDATE products SET stock = stock - (it->>'qty')::int WHERE id = (it->>'product_id')::int;
kafe-#     INSERT INTO order_items VALUES (v_order, (it->>'product_id')::int, (it->>'qty')::int, v_price);
kafe-#   END LOOP;
kafe-#   RETURN QUERY SELECT v_order, sum(oi.qty * oi.unit_price) FROM order_items oi WHERE oi.order_id = v_order;
kafe-# END $$;
CREATE FUNCTION
