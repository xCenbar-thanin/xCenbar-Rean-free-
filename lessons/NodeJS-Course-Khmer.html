១. Node.js ជាអ្វី?

Node.js គឺជា JavaScript Runtime ដែលអនុញ្ញាតឱ្យយើងដំណើរការកូដ JavaScript នៅខាង Server (ក្រៅ Browser)។ វាត្រូវបានសាងសង់លើ V8 Engine របស់ Google Chrome ដែលបំប្លែងកូដ JavaScript ទៅជា Machine Code យ៉ាងលឿន។

មុនពេលមាន Node.js, JavaScript អាចដំណើរការតែនៅក្នុង Browser ប៉ុណ្ណោះ។ ឥឡូវនេះ យើងអាចប្រើភាសាតែមួយ (JavaScript) សម្រាប់ទាំង Frontend និង Backend។

ជំហានទី ១: ដំឡើង Node.js

មេរៀន ២Modules (CommonJS & ES Modules)

Module ជួយបំបែកកូដជា File តូចៗ ងាយស្រួលថែទាំ និងប្រើឡើងវិញ។

ក) CommonJS (របៀបដើម)

math.js
function add(a, b) {
  return a + b;
}
const PI = 3.14159;

module.exports = { add, PI };
main.js
const { add, PI } = require("./math");
const os = require("os"); // Core module

console.log(add(2, 3)); // 5
console.log(PI);

ខ) ES Modules (របៀបទំនើប – ណែនាំ)

បន្ថែម "type": "module" ក្នុង package.json ឬប្តូរឈ្មោះ File ជា .mjs។

math.mjs
export function add(a, b) {
  return a + b;
}
export const PI = 3.14159;

export default function multiply(a, b) {
  return a * b;
}
main.mjs
import multiply, { add, PI } from "./math.mjs";
import os from "node:os"; // prefix "node:" សម្រាប់ core module

console.log(add(2, 3), multiply(4, 5), PI);

✍️ លំហាត់ ២

បង្កើត Module student.js ដែល Export Function getGrade(score) (≥90 → A, ≥80 → B, ≥70 → C, ≥50 → D, ក្រៅពីនេះ → F)។ រួច Import ទៅប្រើក្នុង index.js។

មើលចម្លើយ
student.js
export function getGrade(score) {
  if (score >= 90) return "A";
  if (score >= 80) return "B";
  if (score >= 70) return "C";
  if (score >= 50) return "D";
  return "F";
}
index.js
import { getGrade } from "./student.js";

const scores = [95, 82, 74, 51, 30];
scores.forEach((s) => console.log(`ពិន្ទុ ${s} → ${getGrade(s)}`));

import { EventEmitter } from "node:events";

class BankAccount extends EventEmitter {
  #balance = 0;

  deposit(amount) {
    this.#balance += amount;
    this.emit("deposit", amount, this.#balance);
  }

  withdraw(amount) {
    if (amount > this.#balance) {
      return this.emit("error", new Error("សមតុល្យមិនគ្រប់គ្រាន់"));
    }
    this.#balance -= amount;
    this.emit("withdraw", amount, this.#balance);
    if (this.#balance < 10) this.emit("lowBalance", this.#balance);
  }
}

const acc = new BankAccount();
acc.on("deposit", (a, b) => console.log(`+$${a} → សមតុល្យ $${b}`));
acc.on("withdraw", (a, b) => console.log(`-$${a} → សមតុល្យ $${b}`));
acc.on("lowBalance", (b) => console.log(`⚠️ សមតុល្យទាប: $${b}`));
acc.on("error", (e) => console.log("❌", e.message));

acc.deposit(50);
acc.withdraw(45);
acc.withdraw(100);

មេរៀន ៧Asynchronous Programming & Event Loop

Callback → Promise → async/await

async.js
// 1) Callback (អាចនាំឱ្យមាន "Callback Hell")
function getUserCb(id, callback) {
  setTimeout(() => callback(null, { id, name: "Dara" }), 500);
}
getUserCb(1, (err, user) => console.log("Callback:", user));

// 2) Promise
function getUser(id) {
  return new Promise((resolve, reject) => {
    setTimeout(() => {
      if (id <= 0) reject(new Error("ID មិនត្រឹមត្រូវ"));
      else resolve({ id, name: "Dara" });
    }, 500);
  });
}
getUser(1)
  .then((user) => console.log("Promise:", user))
  .catch((err) => console.error(err.message));

// 3) async/await – អានងាយបំផុត
async function main() {
  try {
    const user = await getUser(2);
    console.log("Await:", user);

    // ដំណើរការស្របគ្នា (Parallel)
    const users = await Promise.all([getUser(3), getUser(4)]);
    console.log("All:", users);
  } catch (err) {
    console.error(err.message);
  }
}
main();

import http from "node:http";

const PORT = 3000;
const products = [
  { id: 1, name: "កាហ្វេ", price: 1.5 },
  { id: 2, name: "តែ", price: 1 },
];

const server = http.createServer((req, res) => {
  const { method, url } = req;

  if (method === "GET" && url === "/") {
    res.writeHead(200, { "Content-Type": "text/html; charset=utf-8" });
    return res.end("<h1>សូមស្វាគមន៍មកកាន់ Node.js Server</h1>");
  }

  if (method === "GET" && url === "/api/products") {
    res.writeHead(200, { "Content-Type": "application/json" });
    return res.end(JSON.stringify(products));
  }

  if (method === "POST" && url === "/api/products") {
    let body = "";
    req.on("data", (chunk) => (body += chunk));
    req.on("end", () => {
      const product = { id: products.length + 1, ...JSON.parse(body) };
      products.push(product);
      res.writeHead(201, { "Content-Type": "application/json" });
      res.end(JSON.stringify(product));
    });
    return;
  }

  res.writeHead(404, { "Content-Type": "application/json" });
  res.end(JSON.stringify({ message: "រកមិនឃើញ" }));
});

server.listen(PORT, () => {
  console.log(`✅ Server កំពុងដំណើរការ http://localhost:${PORT}`);
});

បង្កើត Express App ដែលមាន Route GET /calc/:op/:a/:b (op = add, sub, mul, div) ហើយ Return លទ្ធផលជា JSON។ បើ op មិនត្រឹមត្រូវ Return Status 400។

import express from "express";
const app = express();

const ops = {
  add: (a, b) => a + b,
  sub: (a, b) => a - b,
  mul: (a, b) => a * b,
  div: (a, b) => (b === 0 ? null : a / b),
};

app.get("/calc/:op/:a/:b", (req, res) => {
  const { op } = req.params;
  const a = Number(req.params.a);
  const b = Number(req.params.b);

  if (!ops[op]) return res.status(400).json({ error: "op មិនត្រឹមត្រូវ" });
  if (isNaN(a) || isNaN(b)) return res.status(400).json({ error: "ត្រូវតែជាលេខ" });

  const result = ops[op](a, b);
  if (result === null) return res.status(400).json({ error: "មិនអាចចែកនឹង 0" });

  res.json({ op, a, b, result });
});

app.listen(3000);

import { randomUUID } from "node:crypto";

const app = express();
app.use(express.json());

let books = [
  { id: randomUUID(), title: "រឿងព្រេងខ្មែរ", author: "ស៊ុន", year: 2020 },
];

// READ ALL
app.get("/api/books", (req, res) => res.json(books));

// READ ONE
app.get("/api/books/:id", (req, res) => {
  const book = books.find((b) => b.id === req.params.id);
  if (!book) return res.status(404).json({ message: "រកមិនឃើញសៀវភៅ" });
  res.json(book);
});

// CREATE
app.post("/api/books", (req, res) => {
  const { title, author, year } = req.body;
  if (!title || !author) {
    return res.status(400).json({ message: "title និង author ត្រូវតែមាន" });
  }
  const book = { id: randomUUID(), title, author, year };
  books.push(book);
  res.status(201).json(book);
});

// UPDATE
app.patch("/api/books/:id", (req, res) => {
  const book = books.find((b) => b.id === req.params.id);
  if (!book) return res.status(404).json({ message: "រកមិនឃើញសៀវភៅ" });
  Object.assign(book, req.body);
  res.json(book);
});

// DELETE
app.delete("/api/books/:id", (req, res) => {
  const before = books.length;
  books = books.filter((b) => b.id !== req.params.id);
  if (books.length === before) return res.status(404).json({ message: "រកមិនឃើញ" });
  res.status(204).end();
});

មេរៀន ១១Middleware & Error Handling

Middleware គឺជា Function ដែលដំណើរការ មុន ពេល Request ទៅដល់ Route។ វាមាន Parameters: (req, res, next)។

middleware.js
import express from "express";
const app = express();

// 1) Logger middleware
function logger(req, res, next) {
  const start = Date.now();
  res.on("finish", () => {
    console.log(`${req.method} ${req.url} → ${res.statusCode} (${Date.now() - start}ms)`);
  });
  next(); // បន្តទៅ middleware/route បន្ទាប់
}

// 2) Auth middleware សាមញ្ញ
function requireApiKey(req, res, next) {
  if (req.headers["x-api-key"] !== "secret123") {
    return res.status(401).json({ message: "គ្មានសិទ្ធិ" });
  }
  next();
}

app.use(express.json());
app.use(logger);

app.get("/public", (req, res) => res.json({ ok: true }));
app.get("/private", requireApiKey, (req, res) => res.json({ secret: "🔐" }));

app.get("/crash", async () => {
  throw new Error("មានបញ្ហាអ្វីមួយ!"); // Express 5 ចាប់ async error ដោយស្វ័យប្រវត្តិ
});

// 3) 404 handler

import { createReadStream, statSync } from "node:fs";

app.get("/video", (req, res) => {
  const path = "video.mp4";
  const { size } = statSync(path);
  const range = req.headers.range;

  if (!range) {
    res.writeHead(200, { "Content-Length": size, "Content-Type": "video/mp4" });
    return createReadStream(path).pipe(res);
  }

  const [startStr, endStr] = range.replace("bytes=", "").split("-");
  const start = Number(startStr);
  const end = endStr ? Number(endStr) : Math.min(start + 1_000_000, size - 1);

  res.writeHead(206, {
    "Content-Range": `bytes ${start}-${end}/${size}`,
    "Accept-Ranges": "bytes",
    "Content-Length": end - start + 1,
    "Content-Type": "video/mp4",
  });
  createReadStream(path, { start, end }).pipe(res);
});

import jwt from "jsonwebtoken";
import { User } from "../models/User.js";
import { AppError } from "../utils/AppError.js";
import { env } from "../config/env.js";

const signToken = (user) =>
  jwt.sign({ sub: user.id, role: user.role }, env.jwtSecret, {
    expiresIn: env.jwtExpiresIn,
  });

export async function register({ name, email, password }) {
  const exists = await User.exists({ email });
  if (exists) throw new AppError("Email នេះត្រូវបានប្រើរួចហើយ", 409);
  const user = await User.create({ name, email, password });
  return { user: { id: user.id, name, email }, token: signToken(user) };
}

export async function login({ email, password }) {
  const user = await User.findOne({ email }).select("+password");
  if (!user || !(await user.comparePassword(password))) {
    throw new AppError("Email ឬ Password មិនត្រឹមត្រូវ", 401);
  }
  return { user: { id: user.id, name: user.name, email }, token: signToken(user) };
}
src/middlewares/auth.js
import jwt from "jsonwebtoken";
import { env } from "../config/env.js";
import { AppError } from "../utils/AppError.js";

export function protect(req, res, next) {
  const header = req.headers.authorization || "";
  const token = header.startsWith("Bearer ") ? header.slice(7) : null;
  if (!token) throw new AppError("សូម Login ជាមុនសិន", 401);

  try {
    const payload = jwt.verify(token, env.jwtSecret);
    req.user = { id: payload.sub, role: payload.role };
    next();
  } catch {
    throw new AppError("Token មិនត្រឹមត្រូវ ឬផុតកំណត់", 401);
  }
}

export const restrictTo = (...roles) => (req, res, next) => {
  if (!roles.includes(req.user.role)) throw new AppError("គ្មានសិទ្ធិ", 403);
  next();
};

export async function list(ownerId, { status, page = 1, limit = 10 }) {

  // Mongoose errors
  if (err.name === "CastError") { err.status = 400; err.message = "ID មិនត្រឹមត្រូវ"; }
  if (err.code === 11000) { err.status = 409; err.message = "ទិន្នន័យស្ទួន"; }

  const status = err.status || 500;
  if (status === 500) console.error(err);

  res.status(status).json({
    message: status === 500 && process.env.NODE_ENV === "production"
      ? "Server Error"
      : err.message,
    ...(process.env.NODE_ENV === "development" && { stack: err.stack }),
  });
};
