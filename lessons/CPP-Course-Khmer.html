១. C/C++ ជាអ្វី?

C គឺជាភាសាសរសេរកម្មវិធីកម្រិតទាប-មធ្យម ដែលដំណើរការជិត Hardware និងលឿនខ្លាំង។ C++ គឺជាភាសាដែលពង្រីកពី C ដោយបន្ថែម Class / OOP, Template និង STL ខណៈរក្សាល្បឿនដដែល។

៥. Debugging ក្នុង VS Code

Debugger ជួយឱ្យឃើញតម្លៃអថេរ និងលំដាប់ដំណើរការកម្មវិធីបន្ទាត់ម្តងមួយ។

  1. បើក file .cpp ហើយ ចុចនៅខាងឆ្វេងលេខបន្ទាត់ (ឬ F9) ដាក់ Breakpoint (ចំណុចក្រហម)។
  2. ចុច F5 (ឬ Run → Start Debugging) ជ្រើស C++ (GDB/LLDB) ហើយជ្រើស g++ build and debug active file។ VS Code បង្កើត tasks.json និង launch.json ឱ្យស្វ័យប្រវត្តិ។
  3. កម្មវិធីឈប់នៅ Breakpoint។ មើលតម្លៃអថេរក្នុង Variables, បន្ថែមកន្សោមក្នុង Watch, មើលលំដាប់ហៅអនុគមន៍ក្នុង Call Stack។
  4. ប្រើប៊ូតុងគ្រប់គ្រង៖ F5 បន្ត, F10 Step Over (រំលងចូលអនុគមន៍), F11 Step Into (ចូលក្នុងអនុគមន៍), Shift+F11 Step Out, Shift+F5 បញ្ឈប់។

បើ Debug ដោយដៃ ត្រូវ Compile ជាមួយ -g (បន្ថែមព័ត៌មាន debug)៖

Terminal
g++ -g -Wall -Wextra -std=c++17 main.cpp -o main
gdb ./main

#include <iostream>
#include <stdexcept>
#include <string>
#include <vector>
using namespace std;

template <typename T>
void swapValue(T &a, T &b) {
    T t = a;
    a = b;
    b = t;
}

template <typename T>
class Stack {
    vector<T> data;
public:
    void push(const T &v) { data.push_back(v); }
    void pop() {
        if (data.empty()) throw out_of_range("Stack is empty");
        data.pop_back();
    }
    const T &top() const {
        if (data.empty()) throw out_of_range("Stack is empty");
        return data.back();
    }
    bool empty() const { return data.empty(); }
    size_t size() const { return data.size(); }
};

int main() {
    int x = 1, y = 2;
    swapValue(x, y);
    cout << x << " " << y << endl;          // 2 1

    Stack<string> s;
    s.push("a");
    s.push("b");
    cout << s.top() << " " << s.size() << endl;   // b 2
    s.pop();
    cout << s.top() << endl;                      // a
}

មេរៀន ២០Linked List

Node នីមួយៗមាន data និង pointer ទៅ node បន្ទាប់។ បន្ថែម/លុបនៅដើម ឬនៅកន្លែងដែលមាន pointer រួច បានលឿន O(1) ប៉ុន្តែការចូលដល់ធាតុទី i យឺត O(n) មិនដូច array។

C++
#include <iostream>
using namespace std;

struct Node {
    int data;
    Node *next;
};

void pushFront(Node *&head, int v) {
    Node *n = new Node{v, head};
    head = n;
}

void printList(Node *head) {
    for (Node *p = head; p != nullptr; p = p->next)
        cout << p->data << " -> ";
    cout << "null" << endl;
}

void freeList(Node *&head) {
    while (head) {
        Node *t = head;
        head = head->next;
        delete t;
    }
}

int main() {
    Node *head = nullptr;
    pushFront(head, 3);
    pushFront(head, 2);
    pushFront(head, 1);
    printList(head);     // 1 -> 2 -> 3 -> null
    freeList(head);
}
⚠️ ចំណាំសំខាន់
  • ពិនិត្យ list ទទេ (head == nullptr) មុនចូលប្រើ head->next។
  • ត្រូវ delete node ទាំងអស់ ដើម្បីជៀស memory leak។

✍️ លំហាត់ ២០

  1. បន្ថែមអនុគមន៍ pushBack។
  2. សរសេរអនុគមន៍រាប់ node និងរកតម្លៃក្នុង list។
  3. សរសេរអនុគមន៍ reverse list។
មើលចម្លើយ

#include <iostream>
using namespace std;

struct Node {
    int data;
    Node *next;
};

void pushBack(Node *&head, int v) {
    Node *n = new Node{v, nullptr};
    if (head == nullptr) { head = n; return; }
    Node *p = head;
    while (p->next != nullptr) p = p->next;
    p->next = n;
}

int countNodes(Node *head) {
    int c = 0;
    for (Node *p = head; p; p = p->next) c++;
    return c;
}

bool contains(Node *head, int v) {
    for (Node *p = head; p; p = p->next)
        if (p->data == v) return true;
    return false;
}

void reverseList(Node *&head) {
    Node *prev = nullptr, *cur = head;
    while (cur) {
        Node *next = cur->next;
        cur->next = prev;
        prev = cur;
        cur = next;
    }
    head = prev;
}

void printList(Node *head) {
    for (Node *p = head; p; p = p->next) cout << p->data << " -> ";
    cout << "null" << endl;
}

void freeList(Node *&head) {
    while (head) { Node *t = head; head = head->next; delete t; }
}

int main() {
    Node *head = nullptr;
    for (int v : {1, 2, 3, 4}) pushBack(head, v);
    printList(head);                                     // 1 -> 2 -> 3 -> 4 -> null
    cout << countNodes(head) << " " << contains(head, 3) << endl;   // 4 1
    reverseList(head);
    printList(head);                                     // 4 -> 3 -> 2 -> 1 -> null
    freeList(head);
}

មេរៀន ២១Stack និង Queue

Stack៖ ចូលក្រោយចេញមុន (LIFO)។ Queue៖ ចូលមុនចេញមុន (FIFO)។

C++
#include <iostream>
#include <stack>
#include <queue>
using namespace std;

int main() {
    stack<int> st;
    st.push(1); st.push(2); st.push(3);
    while (!st.empty()) {
        cout << st.top() << " ";   // 3 2 1
        st.pop();
    }
    cout << endl;

    queue<int> q;
    q.push(1); q.push(2); q.push(3);
    while (!q.empty()) {
        cout << q.front() << " ";  // 1 2 3
        q.pop();
    }
}
⚠️ ចំណាំសំខាន់
  • top()/front()/pop() លើ container ទទេ ជា undefined behavior ត្រូវពិនិត្យ empty() មុន។
  • pop() មិនត្រឡប់តម្លៃ ត្រូវ top() មុន ហើយ pop()។

✍️ លំហាត់ ២១

  1. ពិនិត្យវង់ក្រចក ( ) [ ] { } ត្រឹមត្រូវ ឬអត់ ដោយ stack។
  2. ក្លែងធ្វើជួរអតិថិជនធនាគារដោយ queue។

#include <iostream>
#include <queue>
#include <stack>
#include <string>
using namespace std;

bool balanced(const string &s) {
    stack<char> st;
    for (char c : s) {
        if (c == '(' || c == '[' || c == '{') {
            st.push(c);
        } else if (c == ')' || c == ']' || c == '}') {
            if (st.empty()) return false;
            char open = st.top();
            st.pop();
            if ((c == ')' && open != '(') || (c == ']' && open != '[') ||
                (c == '}' && open != '{')) return false;
        }
    }
    return st.empty();
}

int main() {
    cout << balanced("{[()]}") << " " << balanced("([)]") << endl;   // 1 0

    queue<string> line;
    for (string name : {"Dara", "Sophea", "Vanna"}) line.push(name);
    int ticket = 1;
    while (!line.empty()) {
        cout << "Counter serves #" << ticket++ << ": " << line.front() << endl;
        line.pop();
    }
}

មេរៀន ២២Algorithm៖ Binary Search និង Big-O

Binary Search ចែកពាក់កណ្តាលរាល់ជំហាន ល្បឿន O(log n) ប្រៀបធៀប Linear Search O(n)។

C++
#include <iostream>
#include <vector>
using namespace std;

int binarySearch(const vector<int> &a, int target) {
    int lo = 0, hi = (int)a.size() - 1;
    while (lo <= hi) {
        int mid = lo + (hi - lo) / 2;
        if (a[mid] == target) return mid;
        if (a[mid] < target) lo = mid + 1;
        else                 hi = mid - 1;
    }
    return -1;   // រកមិនឃើញ
}

int main() {
    vector<int> a = {1, 3, 5, 7, 9, 11};
    cout << binarySearch(a, 7) << endl;   // 3
    cout << binarySearch(a, 4) << endl;   // -1
}
⚠️ ចំណាំសំខាន់
  • Binary Search ដំណើរការតែលើ array ដែលតម្រៀបរួច។
  • គណនា mid = lo + (hi - lo) / 2 ជៀសវាង overflow។

✍️ លំហាត់ ២២

  1. សរសេរ Binary Search ដោយ recursion។
  2. សរសេរ Selection Sort ហើយប្រៀបធៀបជាមួយ sort()។
មើលចម្លើយ
answer22.cpp
#include <algorithm>
#include <iostream>
#include <vector>
using namespace std;

int bs(const vector<int> &a, int lo, int hi, int t) {
    if (lo > hi) return -1;                 // base case
    int mid = lo + (hi - lo) / 2;
    if (a[mid] == t) return mid;
    if (a[mid] < t) return bs(a, mid + 1, hi, t);
    return bs(a, lo, mid - 1, t);
}

void selectionSort(vector<int> &a) {        // O(n²)
    for (size_t i = 0; i + 1 < a.size(); i++) {
        size_t minIdx = i;
        for (size_t j = i + 1; j < a.size(); j++)
            if (a[j] < a[minIdx]) minIdx = j;
        swap(a[i], a[minIdx]);
    }
}

int main() {
    vector<int> a = {1, 3, 5, 7, 9, 11};
    cout << bs(a, 0, (int)a.size() - 1, 9) << endl;   // 4

    vector<int> v = {5, 2, 9, 1, 7}, w = v;
    selectionSort(v);
    sort(w.begin(), w.end());                // O(n log n) — លឿនជាងសម្រាប់ទិន្នន័យធំ
    cout << (v == w ? "same result" : "different") << endl;
}

មេរៀន ២៣Operator Overloading

កំណត់ឱ្យ +, ==, << ដំណើរការជាមួយ class របស់យើង។

C++
#include <iostream>
using namespace std;

class Vec2 {
public:
    double x, y;
    Vec2(double x = 0, double y = 0) : x(x), y(y) {}

    Vec2 operator+(const Vec2 &o) const {
        return Vec2(x + o.x, y + o.y);
    }
    bool operator==(const Vec2 &o) const {
        return x == o.x && y == o.y;
    }
};

ostream &operator<<(ostream &os, const Vec2 &v) {
    return os << "(" << v.x << ", " << v.y << ")";
}

int main() {
    Vec2 a(1, 2), b(3, 4);
    cout << a + b << endl;          // (4, 6)
    cout << (a == b) << endl;       // 0
}
⚠️ ចំណាំសំខាន់
  • Overload ឱ្យមានន័យធម្មជាតិ (+ សម្រាប់បូក)។ កុំធ្វើឱ្យសាំញ៉ាំ។
  • operator<< ជាអនុគមន៍ខាងក្រៅ class ហើយត្រឡប់ ostream&។

✍️ លំហាត់ ២៣

  1. បន្ថែម operator- និង operator* (គុណនឹងចំនួន) ដល់ Vec2។
  2. បង្កើត class Fraction ហើយ overload + និង <<។
មើលចម្លើយ
answer23.cpp
#include <iostream>
#include <numeric>
#include <stdexcept>
using namespace std;

class Vec2 {
public:
    double x, y;
    Vec2(double x = 0, double y = 0) : x(x), y(y) {}
    Vec2 operator+(const Vec2 &o) const { return Vec2(x + o.x, y + o.y); }
    Vec2 operator-(const Vec2 &o) const { return Vec2(x - o.x, y - o.y); }
    Vec2 operator*(double k) const { return Vec2(x * k, y * k); }
};

ostream &operator<<(ostream &os, const Vec2 &v) {
    return os << "(" << v.x << ", " << v.y << ")";
}

class Fraction {
    long long num, den;
    void reduce() {
        long long g = gcd(num, den);         // C++17 <numeric>
        num /= g; den /= g;
        if (den < 0) { num = -num; den = -den; }
    }
public:
    Fraction(long long n, long long d = 1) : num(n), den(d) {
        if (d == 0) throw invalid_argument("Denominator cannot be zero");
        reduce();
    }
    Fraction operator+(const Fraction &o) const {
        return Fraction(num * o.den + o.num * den, den * o.den);
    }
    friend ostream &operator<<(ostream &os, const Fraction &f) {
        return os << f.num << "/" << f.den;
    }
};

int main() {
    Vec2 a(1, 2), b(3, 4);
    cout << b - a << " " << a * 3 << endl;            // (2, 2) (3, 6)
    cout << Fraction(1, 2) + Fraction(1, 3) << endl;  // 5/6
    cout << Fraction(1, 4) + Fraction(1, 4) << endl;  // 1/2
}

មេរៀន ២៤Multithreading (មូលដ្ឋាន)

Thread ឱ្យកម្មវិធីធ្វើការច្រើនក្នុងពេលតែមួយ។ ប្រើ std::thread និង mutex ការពារទិន្នន័យរួម។

C++
#include <iostream>
#include <thread>
#include <mutex>
using namespace std;

int counter = 0;
mutex m;

void work(int id) {
    for (int i = 0; i < 1000; i++) {
        lock_guard<mutex> lock(m);   // ការពារ data race
        counter++;
    }
    cout << "thread " << id << " done" << endl;
}

int main() {
    thread t1(work, 1);
    thread t2(work, 2);
    t1.join();
    t2.join();
    cout << counter << endl;   // 2000
}
Terminal
g++ -std=c++17 -pthread main.cpp -o app
⚠️ ចំណាំសំខាន់
  • មុនបញ្ចប់ main ត្រូវ join() (ឬ detach()) thread ទាំងអស់។
  • ទិន្នន័យរួមដែលកែក្នុងពេលតែមួយ ត្រូវការ mutex (បើអត់ កើត data race)។
  • Compile ត្រូវបន្ថែម -pthread លើ Linux/macOS/MinGW។

✍️ លំហាត់ ២៤

  1. យកចេញ lock_guard ហើយសង្កេតលទ្ធផល counter (ប្រៀបធៀប)។
  2. ប្រើ ៤ thread គណនាផលបូកនៃ array ធំ ដោយចែកជាផ្នែក។
មើលចម្លើយ

#include <iostream>
#include <numeric>
#include <thread>
#include <vector>
using namespace std;

int main() {
    vector<int> data(1'000'000, 1);              // array ធំ (តម្លៃ 1 ទាំងអស់)
    const int T = 4;
    vector<long long> partial(T, 0);             // thread នីមួយៗសរសេរកន្លែងខ្លួនឯង → គ្មាន data race
    vector<thread> threads;
    size_t chunk = data.size() / T;

    for (int t = 0; t < T; t++) {
        size_t begin = t * chunk;
        size_t end = (t == T - 1) ? data.size() : begin + chunk;
        threads.emplace_back([&, t, begin, end] {
            partial[t] = accumulate(data.begin() + begin, data.begin() + end, 0LL);
        });
    }
    for (auto &th : threads) th.join();

    cout << accumulate(partial.begin(), partial.end(), 0LL) << endl;   // 1000000
}

មេរៀន ២៥Modern C++ — auto, range-for និង optional

Modern C++ គឺជារបៀបសរសេរ C++ សម័យថ្មី ដែលផ្តោតលើសុវត្ថិភាព កូដខ្លី និងងាយថែទាំ។ auto អនុញ្ញាតឱ្យ compiler កំណត់ប្រភេទទិន្នន័យដោយស្វ័យប្រវត្តិ។ range-for ជួយដើរឆ្លងកាត់ធាតុក្នុង container។ optional តំណាងឱ្យតម្លៃដែលអាចមាន ឬមិនមាន។

#include <iostream>
#include <optional>
#include <vector>
using namespace std;

optional<int> findMax(const vector<int> &v) {
    if (v.empty()) return nullopt;
    int best = v[0];
    for (int x : v)
        if (x > best) best = x;
    return best;
}

int main() {

}

មេរៀន ២៧Binary Search Tree (BST)

Tree ជារចនាសម្ព័ន្ធទិន្នន័យដែលមាន node និងការតភ្ជាប់រវាង node។ ក្នុង Binary Search Tree (BST) តម្លៃតូចជាង node បច្ចុប្បន្នស្ថិតនៅខាងឆ្វេង ហើយតម្លៃធំជាងស្ថិតនៅខាងស្តាំ។ ការដើរតាម inorder traversal នឹងបង្ហាញតម្លៃតាមលំដាប់ពីតូចទៅធំ ប្រសិនបើ BST ត្រឹមត្រូវ។

C++
#include <iostream>
using namespace std;

struct Node {
    int value;
    Node* left;
    Node* right;
    Node(int v) : value(v), left(nullptr), right(nullptr) {}
};

void insert(Node*& root, int value) {
    if (root == nullptr) {
        root = new Node(value);
    } else if (value < root->value) {
        insert(root->left, value);
    } else if (value > root->value) {
        insert(root->right, value);
    }
}

void inorder(Node* root) {
    if (root == nullptr) return;
    inorder(root->left);
    cout << root->value << " ";
    inorder(root->right);
}

void destroy(Node* root) {
    if (!root) return;
    destroy(root->left);
    destroy(root->right);
    delete root;
}

int main() {
    Node* root = nullptr;
    insert(root, 8);
    insert(root, 3);
    insert(root, 10);
    insert(root, 1);
    inorder(root); // 1 3 8 10
    destroy(root);
}
⚠️ ចំណាំសំខាន់
  • ត្រូវគ្រប់គ្រង memory និងលុប node ទាំងអស់នៅពេលប្រើ pointer ដោយដៃ។
  • BST ដែលមានតុល្យភាពអាចស្វែងរកបានលឿន; BST ដែលលំអៀងអាចយឺតដល់ O(n)។
  • កូដខាងលើមិនបញ្ចូលតម្លៃស្ទួនទេ។

✍️ លំហាត់ ២៧

  1. បន្ថែមអនុគមន៍ search() ដើម្បីរកលេខក្នុង BST។
  2. បន្ថែមអនុគមន៍រាប់ចំនួន node និងរកតម្លៃតូចបំផុត។
មើលចម្លើយ

#include <iostream>
using namespace std;

struct Node {
    int value;
    Node *left = nullptr, *right = nullptr;
    Node(int v) : value(v) {}
};

void insert(Node *&root, int v) {
    if (!root) root = new Node(v);
    else if (v < root->value) insert(root->left, v);
    else if (v > root->value) insert(root->right, v);
}

bool search(Node *root, int v) {
    while (root) {
        if (v == root->value) return true;
        root = (v < root->value) ? root->left : root->right;
    }
    return false;
}

int countNodes(Node *root) {
    return root ? 1 + countNodes(root->left) + countNodes(root->right) : 0;
}

int minValue(Node *root) {          // សន្មតថា root មិនទទេ
    while (root->left) root = root->left;
    return root->value;
}

void destroy(Node *root) {
    if (!root) return;
    destroy(root->left);
    destroy(root->right);
    delete root;
}

int main() {
    Node *root = nullptr;
    for (int v : {8, 3, 10, 1, 6, 14}) insert(root, v);
    cout << search(root, 6) << " " << search(root, 7) << endl;   // 1 0
    cout << countNodes(root) << " " << minValue(root) << endl;   // 6 1
    destroy(root);
}

មេរៀន ២៨Graph, BFS និង DFS

Graph មាន vertex (កំពូល) និង edge (ខ្សែភ្ជាប់)។ វាអាចតំណាងឱ្យទីក្រុង និងផ្លូវ មិត្តភាព ឬបណ្តាញកុំព្យូទ័រ។ BFS (Breadth-First Search) ស្វែងរកតាមកម្រិតដោយប្រើ queue។ DFS (Depth-First Search) ស្វែងរកជ្រៅដោយប្រើ recursion ឬ stack។

C++
#include <iostream>
#include <vector>
#include <queue>
using namespace std;

int main() {
    vector<vector<int>> graph = {
        {1, 2}, {0, 3}, {0}, {1}
    };
    vector<bool> seen(4, false);
    queue<int> q;

    q.push(0);
    seen[0] = true;

    while (!q.empty()) {
        int u = q.front();
        q.pop();
        cout << u << " ";

        for (int v : graph[u]) {
            if (!seen[v]) {
                seen[v] = true;
                q.push(v);
            }
        }
    }
}
⚠️ ចំណាំសំខាន់
  • ត្រូវប្រើ seen ឬ visited ដើម្បីកុំឱ្យដើរជារង្វង់គ្មានទីបញ្ចប់។
  • BFS អាចរកផ្លូវខ្លីបំផុតក្នុង graph ដែលគ្រប់ edge មានទម្ងន់ស្មើគ្នា។
  • Graph មានទម្ងន់ត្រូវការក្បួនដោះស្រាយផ្សេង ដូចជា Dijkstra។

✍️ លំហាត់ ២៨

  1. បង្កើត graph តំណាងឱ្យទីក្រុង ៥ និងផ្លូវរវាងទីក្រុង។
  2. សរសេរ DFS ហើយប្រៀបធៀបលទ្ធផលជាមួយ BFS។

#include <iostream>
#include <queue>
#include <string>
#include <vector>
using namespace std;

vector<string> city = {"Phnom Penh", "Kampot", "Siem Reap", "Battambang", "Kep"};
vector<vector<int>> road = {
    {1, 2},     // Phnom Penh – Kampot, Siem Reap
    {0, 4},     // Kampot – Phnom Penh, Kep
    {0, 3},     // Siem Reap – Phnom Penh, Battambang
    {2},        // Battambang – Siem Reap
    {1}         // Kep – Kampot
};

void dfs(int u, vector<bool> &seen) {
    seen[u] = true;
    cout << city[u] << " | ";
    for (int v : road[u])
        if (!seen[v]) dfs(v, seen);
}

void bfs(int start) {
    vector<bool> seen(city.size(), false);
    queue<int> q;
    q.push(start);
    seen[start] = true;
    while (!q.empty()) {
        int u = q.front(); q.pop();
        cout << city[u] << " | ";
        for (int v : road[u])
            if (!seen[v]) { seen[v] = true; q.push(v); }
    }
}

}

មេរៀន ២៩Recursion, Dynamic Programming និង Memoization

Recursion គឺអនុគមន៍ហៅខ្លួនឯង។ ត្រូវមាន base case ដើម្បីបញ្ឈប់ការហៅ។ Dynamic Programming (DP) រក្សាទុកចម្លើយរង ដើម្បីកុំគណនាដដែលៗ។ Memoization គឺការរក្សាទុកលទ្ធផលដែលបានគណនារួច។

C++
#include <iostream>
#include <vector>
using namespace std;

long long fibonacci(int n) {
    if (n <= 1) return n;

    vector<long long> dp(n + 1, 0);
    dp[1] = 1;

    for (int i = 2; i <= n; ++i) {
        dp[i] = dp[i - 1] + dp[i - 2];
    }
    return dp[n];
}

int main() {
    cout << fibonacci(10); // 55
}
⚠️ ចំណាំសំខាន់
  • Recursion ត្រូវមាន base case មិនដូច្នោះអាច stack overflow។
  • DP មានប្រយោជន៍ពេលបញ្ហាមាន subproblems ដែលកើតឡើងដដែលៗ។
  • គិតពីទំហំ memory និងព្រំដែន input មុនប្រើ array DP។

✍️ លំហាត់ ២៩

  1. សរសេរ factorial ដោយ recursion និង loop រួចប្រៀបធៀប។
  2. ដោះស្រាយ Fibonacci ដោយ recursion ធម្មតា និង DP រួចប្រៀបធៀបល្បឿន។

#include <iostream>

long long factRec(int n)  { return n <= 1 ? 1 : n * factRec(n - 1); }
long long factLoop(int n) { long long r = 1; for (int i = 2; i <= n; i++) r *= i; return r; }

long long fibRec(int n) { return n <= 1 ? n : fibRec(n - 1) + fibRec(n - 2); }   // O(2^n)

long long fibDP(int n) {                                                          // O(n)
    if (n <= 1) return n;
    vector<long long> dp(n + 1);
    dp[0] = 0; dp[1] = 1;
    for (int i = 2; i <= n; i++) dp[i] = dp[i - 1] + dp[i - 2];
    return dp[n];
}

template <typename F>
void timeIt(const char *name, F f) {
    auto t0 = chrono::steady_clock::now();
    long long r = f();
    auto ms = chrono::duration_cast<chrono::milliseconds>(chrono::steady_clock::now() - t0).count();
    cout << name << " = " << r << " (" << ms << " ms)\n";
}

int main() {

    timeIt("fibRec(35)", [] { return fibRec(35); });      // យឺត

}

ហ្គេមអ្នកលេង ២ នាក់ លើ Console។ ប្រើ array 2 វិមាត្រ, អនុគមន៍ និង loop។ វាយ 1-9 ដើម្បីជ្រើសប្រអប់។ Compile៖ g++ tictactoe.cpp -o ttt

C++
#include <iostream>
using namespace std;

char board[3][3];

void reset() {
    char c = '1';
    for (int r = 0; r < 3; r++)
        for (int k = 0; k < 3; k++)
            board[r][k] = c++;
}

void show() {
    cout << endl;
    for (int r = 0; r < 3; r++) {
        cout << " " << board[r][0] << " | " << board[r][1] << " | " << board[r][2] << endl;
        if (r < 2) cout << "---+---+---" << endl;
    }
    cout << endl;
}

bool place(int pos, char player) {
    if (pos < 1 || pos > 9) return false;
    int r = (pos - 1) / 3, c = (pos - 1) % 3;
    if (board[r][c] == 'X' || board[r][c] == 'O') return false;
    board[r][c] = player;
    return true;
}

bool win(char p) {
    for (int i = 0; i < 3; i++) {
        if (board[i][0] == p && board[i][1] == p && board[i][2] == p) return true;
        if (board[0][i] == p && board[1][i] == p && board[2][i] == p) return true;
    }
    if (board[0][0] == p && board[1][1] == p && board[2][2] == p) return true;
    if (board[0][2] == p && board[1][1] == p && board[2][0] == p) return true;
    return false;
}

int main() {
    reset();
    char player = 'X';
    for (int turn = 0; turn < 9; turn++) {
        show();
        int pos;
        cout << "Player " << player << " (1-9): ";
        if (!(cin >> pos)) return 0;
        if (!place(pos, player)) {
            cout << "Invalid move, try again." << endl;
            turn--;
            continue;
        }
        if (win(player)) {
            show();
            cout << "Player " << player << " wins!" << endl;
            return 0;
        }
        player = (player == 'X') ? 'O' : 'X';
    }
    show();
    cout << "Draw!" << endl;
    return 0;
}

ប្រើ class, vector, exception និង menu។ ក្នុង Account មិនឱ្យដកលើសសមតុល្យ។ Compile៖ g++ -std=c++17 bank.cpp -o bank

C++
#include <iostream>
#include <string>
#include <vector>
#include <stdexcept>
using namespace std;

class Account {
    int id;
    string owner;
    double balance;
public:
    Account(int id, string owner, double balance = 0)
        : id(id), owner(owner), balance(balance) {}

    int getId() const { return id; }
    string getOwner() const { return owner; }
    double getBalance() const { return balance; }

    void deposit(double amount) {
        if (amount <= 0) throw invalid_argument("Amount must be positive");
        balance += amount;
    }
    void withdraw(double amount) {
        if (amount <= 0) throw invalid_argument("Amount must be positive");
        if (amount > balance) throw runtime_error("Insufficient funds");
        balance -= amount;
    }
};

class Bank {
    vector<Account> accounts;
    int nextId = 1001;

    Account *find(int id) {
        for (Account &a : accounts)
            if (a.getId() == id) return &a;
        return nullptr;
    }
public:
    int open(const string &owner, double initial) {
        accounts.push_back(Account(nextId, owner, initial));
        return nextId++;
    }
    void deposit(int id, double amt) {
        Account *a = find(id);
        if (!a) throw runtime_error("Account not found");
        a->deposit(amt);
    }
    void withdraw(int id, double amt) {
        Account *a = find(id);
        if (!a) throw runtime_error("Account not found");
        a->withdraw(amt);
    }
    void list() const {
        for (const Account &a : accounts)
            cout << a.getId() << "  " << a.getOwner() << "  $" << a.getBalance() << endl;
    }
};

int main() {
    Bank bank;
    int choice;
    do {
        cout << "\n1.Open  2.Deposit  3.Withdraw  4.List  0.Exit\n> ";
        cin >> choice;
        try {
            if (choice == 1) {
                string name; double init;
                cout << "Name: "; cin >> name;
                cout << "Initial: "; cin >> init;
                cout << "Account ID: " << bank.open(name, init) << endl;
            } else if (choice == 2 || choice == 3) {
                int id; double amt;
                cout << "ID: "; cin >> id;
                cout << "Amount: "; cin >> amt;
                if (choice == 2) bank.deposit(id, amt);
                else bank.withdraw(id, amt);
                cout << "Done." << endl;
            } else if (choice == 4) {
                bank.list();
            }
        } catch (const exception &e) {
            cout << "Error: " << e.what() << endl;
        }
    } while (choice != 0 && cin);
    return 0;
}

#include "Money.h"

#include <cctype>
#include <stdexcept>

Cents parseMoney(const std::string &text) {
    Cents whole = 0, frac = 0;
    int fracDigits = 0;
    bool seenDot = false, seenDigit = false;
    for (char c : text) {
        if (c == '.' && !seenDot) { seenDot = true; continue; }
        if (!std::isdigit(static_cast<unsigned char>(c))) throw std::invalid_argument("Invalid amount: " + text);
        seenDigit = true;
        if (!seenDot) {
            whole = whole * 10 + (c - '0');
            if (whole > 1'000'000'000) throw std::invalid_argument("Amount too large");
        } else if (++fracDigits <= 2) {
            frac = frac * 10 + (c - '0');
        } else {
            throw std::invalid_argument("Max 2 decimal places");
        }
    }
    if (!seenDigit) throw std::invalid_argument("Invalid amount: " + text);
    if (fracDigits == 1) frac *= 10;
    return whole * 100 + frac;
}

std::string formatMoney(Cents value) {
    std::string sign = value < 0 ? "-" : "";
    if (value < 0) value = -value;
    std::string cents = std::to_string(value % 100);
    if (cents.size() == 1) cents = "0" + cents;
    return sign + std::to_string(value / 100) + "." + cents;
}

Account::Account(int id, std::string owner, Cents initial)
    : id_(id), owner_(std::move(owner)), balance_(0) {
    if (owner_.empty()) throw std::invalid_argument("Owner name is required");
    if (initial < 0) throw std::invalid_argument("Initial balance cannot be negative");
    if (initial > 0) deposit(initial);
}

void Account::deposit(Cents amount, TxType type) {
    if (amount <= 0) throw std::invalid_argument("Amount must be positive");
    balance_ += amount;
    history_.push_back({type, amount, balance_});
}

void Account::withdraw(Cents amount, TxType type) {
    if (amount <= 0) throw std::invalid_argument("Amount must be positive");
    if (amount > balance_) throw std::runtime_error("Insufficient funds");
    balance_ -= amount;
    history_.push_back({type, amount, balance_});
}

const char *toString(TxType type) {
    switch (type) {
        case TxType::Deposit:     return "Deposit";
        case TxType::Withdraw:    return "Withdraw";
        case TxType::TransferIn:  return "Transfer in";
        case TxType::TransferOut: return "Transfer out";
    }
    return "?";
}

#include "Bank.h"

#include <cstdio>
#include <fstream>
#include <sstream>
#include <stdexcept>

int Bank::open(const std::string &owner, Cents initial) {
    accounts_.emplace_back(nextId_, owner, initial);
    return nextId_++;
}

Account &Bank::find(int id) {
    for (Account &a : accounts_)
        if (a.id() == id) return a;
    throw std::runtime_error("Account " + std::to_string(id) + " not found");
}

const Account &Bank::get(int id) const {
    for (const Account &a : accounts_)
        if (a.id() == id) return a;
    throw std::runtime_error("Account " + std::to_string(id) + " not found");
}

void Bank::deposit(int id, Cents amount) { find(id).deposit(amount); }
void Bank::withdraw(int id, Cents amount) { find(id).withdraw(amount); }

void Bank::transfer(int fromId, int toId, Cents amount) {
    if (fromId == toId) throw std::invalid_argument("Cannot transfer to the same account");
    Account &from = find(fromId);
    Account &to = find(toId);                 // ពិនិត្យថាទាំងពីរមាន មុនកែប្រែអ្វីទាំងអស់
    from.withdraw(amount, TxType::TransferOut);
    to.deposit(amount, TxType::TransferIn);
}

// ទម្រង់ file (CSV): id,owner,balance_cents  — ឈ្មោះមិនអនុញ្ញាតឱ្យមាន ','
void Bank::save(const std::string &path) const {
    const std::string tmp = path + ".tmp";
    {
        std::ofstream out(tmp);
        if (!out) throw std::runtime_error("Cannot write " + tmp);
        out << nextId_ << "\n";
        for (const Account &a : accounts_)
            out << a.id() << "," << a.owner() << "," << a.balance() << "\n";
        if (!out) throw std::runtime_error("Write failed");
    }
    std::remove(path.c_str());                // សរសេរ file ថ្មីរួចហើយ ទើបជំនួស (ការពារ file ខូចពាក់កណ្តាល)
    if (std::rename(tmp.c_str(), path.c_str()) != 0) throw std::runtime_error("Cannot save " + path);
}

void Bank::load(const std::string &path) {
    std::ifstream in(path);
    if (!in) return;                          // គ្មាន file = ចាប់ផ្តើមថ្មី
    std::vector<Account> loaded;
    int nextId = 1001;
    std::string line;
    if (std::getline(in, line)) nextId = std::stoi(line);
    while (std::getline(in, line)) {
        std::stringstream ss(line);
        std::string id, owner, balance;
        if (!std::getline(ss, id, ',') || !std::getline(ss, owner, ',') || !std::getline(ss, balance))
            throw std::runtime_error("Corrupted line: " + line);
        loaded.emplace_back(std::stoi(id), owner, std::stoll(balance));
    }
    accounts_ = std::move(loaded);
    nextId_ = nextId;
}

#include <iostream>
#include <limits>
#include <string>

#include "Bank.h"

namespace {

const std::string DATA_FILE = "bank_data.csv";

std::string ask(const std::string &prompt) {
    std::cout << prompt;
    std::string s;
    if (!std::getline(std::cin, s)) throw std::runtime_error("Input closed");
    return s;
}

int askInt(const std::string &prompt) {
    while (true) {
        std::string s = ask(prompt);
        try {
            size_t used = 0;
            int v = std::stoi(s, &used);
            if (used == s.size()) return v;
        } catch (const std::exception &) {}
        std::cout << "  Please enter a whole number.\n";
    }
}

Cents askMoney(const std::string &prompt) {
    while (true) {
        try { return parseMoney(ask(prompt)); }
        catch (const std::invalid_argument &e) { std::cout << "  " << e.what() << "\n"; }
    }
}

void showHistory(const Account &a) {
    std::cout << "History of " << a.id() << " (" << a.owner() << ")\n";
    for (const Transaction &t : a.history())
        std::cout << "  " << toString(t.type) << "  $" << formatMoney(t.amount)
                  << "  → balance $" << formatMoney(t.balanceAfter) << "\n";
}

}  // namespace

int main() {
    Bank bank;
    try {
        bank.load(DATA_FILE);
    } catch (const std::exception &e) {
        std::cerr << "Could not load data: " << e.what() << "\n";
        return 1;
    }

    while (true) {
        std::cout << "\n1.Open 2.Deposit 3.Withdraw 4.Transfer 5.Balance 6.History 7.List 0.Save & Exit\n";
        try {
            int choice = askInt("> ");
            switch (choice) {
                case 1: {
                    std::string name = ask("Owner name: ");
                    if (name.find(',') != std::string::npos) throw std::invalid_argument("Name cannot contain ','");
                    int id = bank.open(name, askMoney("Initial deposit: "));
                    std::cout << "Account created: " << id << "\n";
                    break;
                }
                case 2: { int id = askInt("Account ID: "); bank.deposit(id, askMoney("Amount: ")); std::cout << "Done.\n"; break; }
                case 3: { int id = askInt("Account ID: "); bank.withdraw(id, askMoney("Amount: ")); std::cout << "Done.\n"; break; }
                case 4: {
                    int from = askInt("From ID: ");
                    int to = askInt("To ID: ");
                    bank.transfer(from, to, askMoney("Amount: "));
                    std::cout << "Transferred.\n";
                    break;
                }

                    std::cout << "Choose 0-7\n";

        } catch (const std::runtime_error &e) {
            if (std::string(e.what()) == "Input closed") { bank.save(DATA_FILE); return 0; }
            std::cout << "Error: " << e.what() << "\n";
        } catch (const std::exception &e) {
            std::cout << "Error: " << e.what() << "\n";
        }
    }
